def fake_sha256(n):
    n += 99991
    part1 = n * 1234567
    part2 = n ** 3
    part3 = n // 7  # Integer division: drops the decimal
    result = (part1 + part2 + part3) % 100000000  # Keep last 8 digits
    return f'PT1: {part1},\\nPT2: {part2},\\nPT3: {part3},\\n(part1 + part2 + part3): {part1 + part2 + part3},\\nInput: {n},\\nANS: {result}'
